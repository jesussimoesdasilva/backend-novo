function toast(msg, type='success') {

async function initLogin() {
  try {
    const r = await fetch(API + '/igrejas');
    const lista = await r.json();
    const ig = lista.find(i => i.slug === slug);
    if (!ig) return document.getElementById('loginNome').textContent = 'Igreja não encontrada';
    
    document.getElementById('loginNome').textContent = ig.nome;
    if (ig.icone && ig.icone.startsWith('/logos/')) {
      document.getElementById('loginLogo').innerHTML = `<img src="${ig.icone}">`;
    } else {
      document.getElementById('loginLogo').textContent = ig.icone || '⛪';
    }
    const saved = localStorage.getItem('subadmin_' + slug);
    if (saved) {
      igrejaAtual = JSON.parse(saved);
      mostrarPainel();
    }
  } catch (e) { console.error(e); }
}

async function fazerLogin(e) {
  e.preventDefault();
  const email = document.getElementById('loginEmail').value.trim();
  const senha = document.getElementById('loginSenha').value;
  const err = document.getElementById('loginError');
  err.classList.remove('show');
  try {
    const r = await fetch(API + '/igrejas/login', {
      method: 'POST', headers: {'Content-Type':'application/json'},
      body: JSON.stringify({ email, senha })
    });
    const d = await r.json();
    if (!r.ok) throw new Error(d.message || 'Erro no login');
    if (d.igreja.slug !== slug) throw new Error('Login não pertence a esta igreja');
    igrejaAtual = d.igreja;
    localStorage.setItem('subadmin_' + slug, JSON.stringify(d.igreja));
    mostrarPainel();
  } catch (e) {
    err.textContent = '❌ ' + e.message;
    err.classList.add('show');
  }
}

async function criarAlbum() {
  const nome = document.getElementById('gaNome').value.trim();
  if (!nome) return toast('Nome do álbum obrigatório', 'error');
  
  try {
    const r = await fetch(API + '/galeria/albuns/igreja/' + igrejaAtual.id, {
      method: 'POST', headers: {'Content-Type':'application/json'},
      body: JSON.stringify({
        nome,
        descricao: document.getElementById('gaDescricao').value,
        dataEvento: document.getElementById('gaData').value || null
      })
    });
    if (!r.ok) throw new Error('Erro ao criar álbum');
    const album = await r.json();

    const file = document.getElementById('gaCapa').files[0];
    if (file) {
      const fd = new FormData(); fd.append('capa', file);
      await fetch(API + '/galeria/albuns/' + album.id + '/capa', { method: 'POST', body: fd });
    }

    toast('Álbum criado!', 'success');
    ['gaNome','gaDescricao','gaData','gaCapa'].forEach(id => document.getElementById(id).value = '');
    carregarAlbuns();
  } catch (e) { toast(e.message, 'error'); }
}

async function carregarOfertas() {
  try {
    const r = await fetch(API + '/ofertas/igreja/' + igrejaAtual.id);
    const lista = await r.json();
    const div = document.getElementById('listaOfertas');
    
    if (!lista.length) return div.innerHTML = '<div class="empty">Nenhuma oferta registada.</div>';
    
    div.innerHTML = `
      <table>
        <thead><tr><th>Código</th><th>Doador</th><th>Finalidade</th><th>Valor</th><th>Estado</th><th>Ações</th></tr></thead>
        <tbody>
          ${lista.map(o => {
            let bgColor = o.status === 'VALIDADO' ? 'rgba(16,185,129,0.1)' : (o.status === 'AGUARDANDO_PASTOR' ? 'rgba(250,204,21,0.1)' : 'transparent');
            let statusTxt = o.status === 'VALIDADO' ? '✅ Validado' : (o.status === 'AGUARDANDO_PASTOR' ? '⏳ Aguarda Pastor' : (o.status === 'NEGADO' ? '❌ Negado' : '🕒 Pendente'));
            
            let acoes = '';
            if (o.status === 'PENDENTE') {
              if (o.finalidade === 'Dízimo') {
                acoes = `<button class="btn btn-primary" style="padding:4px 8px;font-size:11px" onclick="preValidar(${o.id})">Pré-Validar</button>`;
              } else {
                acoes = `<button class="btn btn-success" style="padding:4px 8px;font-size:11px" onclick="validarOferta(${o.id})">Validar</button>`;
              }
              acoes += ` <button class="btn btn-danger" style="padding:4px 8px;font-size:11px" onclick="negarOferta(${o.id})">Negar</button>`;
            }
            
            return `
              <tr style="background:${bgColor}">
                <td style="font-family:monospace">${o.codigo}</td>
                <td>${o.membro ? o.membro.nome : (o.nomeDoador || 'Anónimo')}</td>
                <td>${o.finalidade}</td>
                <td><strong>${o.valor.toLocaleString('pt-AO')} Kz</strong></td>
                <td>${statusTxt}</td>
                <td>${acoes}</td>
              </tr>
            `;
          }).join('')}
        </tbody>
      </table>
    `;
  } catch (e) { document.getElementById('listaOfertas').innerHTML = '<div class="empty" style="color:var(--red)">Erro ao carregar ofertas</div>'; }
}

async function salvarPastor() {
  const nome = document.getElementById('pNome').value.trim();
  if (!nome) return toast('Nome obrigatório', 'error');
  
  const payload = {
    nome, cargo: document.getElementById('pCargo').value, telefone: document.getElementById('pTelefone').value,
    email: document.getElementById('pEmail').value, versiculo: document.getElementById('pVersiculo').value,
    bio: document.getElementById('pBio').value, conselho: document.getElementById('pConselho').value
  };

  try {
    let url = API + '/pastores/igreja/' + igrejaAtual.id;
    let method = 'POST';

    if (pastorEditandoId) {
      url = API + '/pastores/' + pastorEditandoId;
      method = 'PUT';
    }

    const r = await fetch(url, {
      method: method, headers: {'Content-Type':'application/json'},
      body: JSON.stringify(payload)
    });
    const d = await r.json();
    if (!r.ok) throw new Error(d.message);

    const file = document.getElementById('pFoto').files[0];
    if (file) {
      const fd = new FormData(); fd.append('foto', file);
      await fetch(API + '/pastores/igreja/' + igrejaAtual.id + '/' + (pastorEditandoId || d.id) + '/foto', { method: 'POST', body: fd });
    }

    toast(pastorEditandoId ? 'Pastor atualizado!' : 'Pastor adicionado!', 'success');
    cancelarEdicaoPastor();
    carregarPastores();
  } catch (e) { toast(e.message, 'error'); }
}

async function editarPastor(id) {
  try {
    const r = await fetch(API + '/pastores/' + id);
    const p = await r.json();

    document.getElementById('pNome').value = p.nome || '';
    document.getElementById('pCargo').value = p.cargo || '';
    document.getElementById('pTelefone').value = p.telefone || '';
    document.getElementById('pEmail').value = p.email || '';
    document.getElementById('pVersiculo').value = p.versiculo || '';
    document.getElementById('pBio').value = p.bio || '';
    document.getElementById('pConselho').value = p.conselho || '';

    pastorEditandoId = id;
    document.getElementById('formPastorTitulo').textContent = '✏️ Editar Pastor';
    document.getElementById('btnSalvarPastor').textContent = '💾 Guardar Alterações';
    document.getElementById('btnCancelarPastor').style.display = 'block';
  } catch (e) { toast('Erro ao carregar os dados do pastor', 'error'); }
}

function cancelarEdicaoPastor() {
  pastorEditandoId = null;
  document.getElementById('formPastorTitulo').textContent = '➕ Adicionar Pastor';
  document.getElementById('btnSalvarPastor').textContent = '💾 Adicionar Pastor';
  document.getElementById('btnCancelarPastor').style.display = 'none';
  ['pNome','pCargo','pTelefone','pEmail','pVersiculo','pBio','pConselho'].forEach(id => document.getElementById(id).value = '');
  document.getElementById('pFoto').value = '';
}

    const div = document.getElementById('listaAvisos');
    
    if (!lista || lista.length === 0) {
      div.innerHTML = '<div class="empty">Nenhum aviso publicado.</div>';
      return;
    }
    
    div.innerHTML = lista.map(a => `
      <div class="depto-item" style="display:flex;justify-content:space-between;align-items:center">
        <div>
          <strong style="color:var(--gold);font-size:16px">${a.titulo}</strong>
          <p style="color:var(--muted);font-size:12px;margin-top:4px">${a.texto}</p>
        </div>
        <button class="btn btn-danger" onclick="apagarAviso(${a.id})">🗑️</button>
      </div>
    `).join('');
  } catch (e) { 
    document.getElementById('listaAvisos').innerHTML = '<div class="empty" style="color:var(--red)">Erro ao carregar avisos</div>'; 
  }
}
async function adicionarAviso() {
  const titulo = document.getElementById('avTitulo').value.trim();
  const texto = document.getElementById('avTexto').value.trim();
  if (!titulo || !texto) return toast('Título e texto são obrigatórios', 'error');
  
  try {
    const r = await fetch(API + '/avisos/igreja/' + igrejaAtual.id, {
      method: 'POST', headers: {'Content-Type':'application/json'},
      body: JSON.stringify({ titulo, texto })
    });
    if (!r.ok) throw new Error('Erro');
    toast('Aviso publicado!', 'success');
    document.getElementById('avTitulo').value = '';
    document.getElementById('avTexto').value = '';
    carregarAvisos();
  } catch (e) { toast(e.message, 'error'); }
}

async function apagarAviso(id) {
  if (!confirm('Apagar este aviso?')) return;
  await fetch(API + '/avisos/' + id, { method: 'DELETE' });
  toast('Aviso apagado', 'success'); carregarAvisos();
}

async function carregarOpcoesChat() {
  const tipo = document.getElementById('chatTipo').value;
  const sel = document.getElementById('chatAlvoId');
  
  if (!tipo) {
    sel.innerHTML = '<option value="">Escolha o tipo primeiro</option>';

    return;
  }

    sel.disabled = false;
  } else {
    try {
      const r = await fetch(API + '/sectores/igreja/' + igrejaAtual.id);
      const sectores = await r.json();
      sel.innerHTML = sectores.map(s => `<option value="${s.id}">${s.nome}</option>`).join('');
      sel.disabled = false;
    } catch (e) { toast('Erro ao carregar sectores', 'error'); }
  }
}

async function carregarMensagensModeracao() {
  const tipo = document.getElementById('chatTipo').value;
  const alvoId = document.getElementById('chatAlvoId').value;
  const div = document.getElementById('listaMensagensModeracao');

  if (!tipo || !alvoId) return toast('Selecione o tipo e o grupo', 'error');
  
  div.innerHTML = 'A carregar mensagens...';
  
  try {
    const url = tipo === 'depto' 
      ? `/api/grupos/mensagens/depto/${alvoId}` 
      : `/api/grupos/mensagens/sector/${alvoId}`;
      
    const r = await fetch(url);
    const msgs = await r.json();
    
    if (!Array.isArray(msgs)) {
       div.innerHTML = `<p style="color:var(--red)">${msgs.message || 'Erro ao carregar.'}</p>`;
       return;
    }

    if (msgs.length === 0) {
      div.innerHTML = '<div class="empty">Nenhuma mensagem neste grupo.</div>';
      return;
    }

    div.innerHTML = msgs.map(m => {
      const hora = new Date(m.criadoEm).toLocaleString('pt-PT');
      return `
        <div class="chat-msg-item">
          <button class="btn btn-danger btn-delete" onclick="apagarMensagem(${m.id})">🗑️ Apagar</button>
          <span class="autor">${m.membro.nome}</span>
          <p class="texto">${m.texto}</p>
          <span class="hora">${hora}</span>
        </div>
      `;
    }).join('');

  } catch (e) {
    div.innerHTML = '<p style="color:var(--red)">Erro de ligação.</p>';
  }
}

async function apagarMensagem(msgId) {
  if (!confirm('Tem a certeza que deseja APAGAR esta mensagem? (Ficará oculta para todos)')) return;
  try {
    const r = await fetch(API + '/grupos/mensagens/' + msgId + '/silenciar', { method: 'PUT' });
    if (!r.ok) throw new Error('Erro ao apagar');
    toast('Mensagem apagada!', 'success');
    carregarMensagensModeracao(); 
  } catch (e) { toast(e.message, 'error'); }
}

}

  } catch (e) { toast(e.message, 'error'); }
}
